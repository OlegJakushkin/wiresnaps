cmake_minimum_required(VERSION 3.20)
project(wiresnaps LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_C_STANDARD 11)
option(WS_SANITIZE "Enable native address and undefined behavior sanitizers" OFF)
if(EMSCRIPTEN)
  add_compile_options(-fexceptions)
  add_link_options(-sDISABLE_EXCEPTION_CATCHING=0)
endif()
if(WS_SANITIZE AND NOT EMSCRIPTEN)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
file(GLOB_RECURSE CORE_SOURCES CONFIGURE_DEPENDS engine/derived/*.cpp)
add_library(simulide_core STATIC ${CORE_SOURCES} engine/src/simulator.cpp engine/src/expression.cpp)
target_include_directories(simulide_core PUBLIC engine/include engine/derived
  engine/derived/elements engine/derived/elements/active engine/derived/elements/passive
  engine/derived/elements/logic engine/derived/elements/outputs)
target_compile_options(simulide_core PRIVATE -Wall -Wextra -Wno-unused-parameter -Wno-sign-compare)
enable_testing()
file(GLOB AVR_SOURCES CONFIGURE_DEPENDS engine/mcu/simavr/sim/*.c engine/mcu/simavr/cores/*.c)
list(FILTER AVR_SOURCES EXCLUDE REGEX "/(sim_hex|sim_vcd_file)\\.c$")
add_library(simavr STATIC ${AVR_SOURCES} engine/src/avr_no_debug.c)
target_include_directories(simavr PUBLIC engine/mcu/simavr engine/mcu/simavr/sim engine/mcu/simavr/cores)
target_compile_definitions(simavr PRIVATE _DEFAULT_SOURCE)
file(GLOB_RECURSE PIC_SOURCES CONFIGURE_DEPENDS engine/mcu/gpsim/*.cc)
add_library(gpsim STATIC ${PIC_SOURCES} engine/src/pic_context.cpp)
target_include_directories(gpsim PUBLIC engine/include engine/mcu/gpsim engine/mcu/gpsim/modules engine/mcu/gpsim/devices engine/mcu/gpsim/registers)
add_library(wiresnaps_mcu STATIC engine/src/mcu_avr.cpp engine/src/mcu_pic.cpp engine/src/hex.cpp)
target_link_libraries(wiresnaps_mcu PUBLIC simavr gpsim)
add_library(wiresnaps_engine STATIC engine/src/engine.cpp engine/src/api.cpp)
target_include_directories(wiresnaps_engine PUBLIC engine/vendor)
target_link_libraries(wiresnaps_engine PUBLIC simulide_core wiresnaps_mcu)
if(NOT EMSCRIPTEN)
  set(CMAKE_POSITION_INDEPENDENT_CODE ON)
  set_target_properties(simulide_core simavr gpsim wiresnaps_mcu wiresnaps_engine PROPERTIES POSITION_INDEPENDENT_CODE ON)
  add_library(wiresnaps SHARED engine/src/api.cpp)
  target_link_libraries(wiresnaps PRIVATE wiresnaps_engine)
endif()
if(EMSCRIPTEN)
  add_executable(wiresnaps engine/src/wasm_entry.cpp)
  target_link_libraries(wiresnaps PRIVATE wiresnaps_engine)
  set_target_properties(wiresnaps PROPERTIES SUFFIX ".mjs")
  target_link_options(wiresnaps PRIVATE --no-entry -sMODULARIZE=1 -sEXPORT_ES6=1 -sEXPORT_NAME=createWireSnaps
    -sALLOW_MEMORY_GROWTH=1 -sMAXIMUM_MEMORY=268435456 -sENVIRONMENT=web,worker,node -sDISABLE_EXCEPTION_CATCHING=0
    "-sEXPORTED_FUNCTIONS=['_ws_api_version','_ws_create','_ws_destroy','_ws_load','_ws_control','_ws_advance','_ws_load_firmware','_ws_reset','_ws_repair','_ws_snapshot','_malloc','_free']"
    "-sEXPORTED_RUNTIME_METHODS=['ccall','cwrap','UTF8ToString']")
  add_compile_options(-fexceptions)
endif()
add_executable(physics_tests engine/tests/physics_tests.cpp)
target_include_directories(physics_tests PRIVATE build/generated)
target_link_libraries(physics_tests PRIVATE wiresnaps_engine)
add_executable(mcu_tests engine/tests/mcu_tests.cpp)
target_include_directories(mcu_tests PRIVATE build/generated)
target_link_libraries(mcu_tests PRIVATE wiresnaps_mcu)
add_executable(core_tests engine/tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE simulide_core)
if(WS_SANITIZE AND NOT EMSCRIPTEN)
  # Keep sanitizer test address maps stable under Docker/WSL ASLR. Production
  # libraries remain PIC; every address/undefined-behavior check stays enabled.
  foreach(test_target core_tests mcu_tests physics_tests)
    target_link_options(${test_target} PRIVATE -no-pie)
  endforeach()
endif()
if(EMSCRIPTEN)
  foreach(test_target core_tests mcu_tests physics_tests)
    set_target_properties(${test_target} PROPERTIES SUFFIX ".mjs")
    target_link_options(${test_target} PRIVATE -sMODULARIZE=1 -sEXPORT_ES6=1)
  endforeach()
  target_link_options(core_tests PRIVATE -sALLOW_MEMORY_GROWTH=1 -sASSERTIONS=1 -sEXIT_RUNTIME=1)
  add_test(NAME core COMMAND node ${CMAKE_SOURCE_DIR}/tools/run_wasm_test.mjs $<TARGET_FILE:core_tests>)
  target_link_options(mcu_tests PRIVATE -sALLOW_MEMORY_GROWTH=1 -sASSERTIONS=1 -sEXIT_RUNTIME=1)
  add_test(NAME mcu COMMAND node ${CMAKE_SOURCE_DIR}/tools/run_wasm_test.mjs $<TARGET_FILE:mcu_tests>)
  target_link_options(physics_tests PRIVATE -sALLOW_MEMORY_GROWTH=1 -sASSERTIONS=1 -sEXIT_RUNTIME=1)
  add_test(NAME physics COMMAND node ${CMAKE_SOURCE_DIR}/tools/run_wasm_test.mjs $<TARGET_FILE:physics_tests>)
else()
  add_test(NAME core COMMAND core_tests)
  add_test(NAME mcu COMMAND mcu_tests)
  add_test(NAME physics COMMAND physics_tests)
endif()
set_tests_properties(core mcu physics PROPERTIES TIMEOUT 90 ENVIRONMENT "UBSAN_OPTIONS=halt_on_error=1:print_stacktrace=1")
